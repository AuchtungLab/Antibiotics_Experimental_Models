#Batch files for analysis of processing for data in mothur can be found in MBRA_Analysis_Files/MBRA_Mothur_Files and Mice_Analysis_Files/MICE_MOTHUR_FILES. Note that input file name, ".files" have been manually adjusted from "CDC.files" to avoid confusion between primarily mouse and exclusively MBRA analyses.
#For MBRA, 
#Start with biom file that was processed on high performance computing cluster and convert to shared and cons.taoxnomy file using mothur 1.48.1
biom.info(biom=CDC.final.abund.opti_mcc.BR.0.01.biom, label=0.01)
#Then, remove samples from a separate study that are not analyzed here.
get.groups(shared=CDC.final.abund.opti_mcc.BR.0.01.0.01.shared, cons.taxonomy=CDC.final.abund.opti_mcc.BR.0.01.0.01.cons.taxonomy, accnos=CDC2.accnos)
#For both MBRA and mice, subsample in mothur using the following code (with the shared files "CDC.final.abund.opti_mcc.shared" for mice and "CDC.final.abund.opti_mcc.BR.0.01.pick.shared" for MBRA):
set.seed(seed=478931)
sub.sample(shared=current, constaxonomy=current, size=6944)
dist.shared(shared=current, calc=braycurtis-jclass, output=square)
#For both MBRA and mice, add column to cons.taxonomy file to add "OTU" as "Species" for analysis at Species-level by ANOCOM-BC2.
#For MBRA, use mothur version 1.44.1 to make biom for use in phyloseq; shared and cons.taxonomy used for mouse data:
make.biom(shared=CDC.final.abund.opti_mcc.BR.0.01.pick.0.01.subsample.shared, constaxonomy=CDC.final.abund.opti_mcc.BR.0.01.0.01.cons.subsample.OTU.cons.taxonomy)

#Import file for analysis

rownames(meta)<-meta$SampleID
BR.phyloseq<-phyloseq(otu_table(ps), tax_table(ps), sample_data(meta))
#Determine alpha diversity
adiv <- estimate_richness(BR.phyloseq)
write.table(adiv, "/Users/jenniferauchtung/Desktop/BR.alphadivvalues.txt", sep="\t")
#Open file in excel and add to existing metadata

#Identify differentially abundant OTUs and Genera
tax_fix_interactive(BR.phyloseq) #Find classifications that are not unique
#Use code to change "uncultured" to unique identifier with next highest classification
BR.phyloseq<-BR.phyloseq %>%
     tax_fix(
         min_length = 4,
         unknowns = c("uncultured"),
         sep = " ", anon_unique = TRUE,
         suffix_rank = "classified")
BR.AA<-subset_samples(BR.phyloseq, Time=="B") #Select just samples after antibiotics
#Write out OTU table of selected data
BR.AA.OTU<-as(otu_table(BR.AA), "matrix")

#Write out Tax table of selected data
BR.AA.TAX<-as(tax_table(BR.AA), "matrix")
write.table(BR.AA.TAX, "/Users/jenniferauchtung/Desktop/CDC/BR.AA.OTU.TAX.table")
#Aggregate taxa by Genus and write out OTU and Tax tables
BR.AA.Genus<-tax_glom(BR.AA, taxrank="Genus")
